# CMake build for the CoDOL fork of iw3xo-radiant.
# Mirrors premake5.lua (which is kept for upstream parity) so the project can be
# built with nothing but CMake + Visual Studio:
#
#   cmake -S . -B build-cmake -G "Visual Studio 17 2022" -A Win32
#   cmake --build build-cmake --config Release
#
# The radiant host process is 32-bit, so only Win32 is supported.

cmake_minimum_required(VERSION 3.20)
project(iw3xo-radiant-codol LANGUAGES C CXX)

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
	message(FATAL_ERROR "iw3r.dll is injected into a 32-bit radiant; configure with -A Win32")
endif()

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "" FORCE)
set(DEPS ${CMAKE_SOURCE_DIR}/deps)

# ---------------------------------------------------------------------------
# version.h / version.hpp (premake's "generate-buildinfo" action)

find_package(Git QUIET)
function(git_out var)
	set(${var} "" PARENT_SCOPE)
	if(GIT_FOUND)
		execute_process(COMMAND ${GIT_EXECUTABLE} ${ARGN}
			WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
			OUTPUT_VARIABLE out OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
		set(${var} "${out}" PARENT_SCOPE)
	endif()
endfunction()

git_out(GIT_DESCRIBE describe --tags --always)
git_out(GIT_HASH rev-parse HEAD)
git_out(GIT_TAG describe --tags --abbrev=0)
git_out(GIT_BRANCH branch --show-current)
git_out(GIT_REVCOUNT rev-list --count HEAD)
if(NOT GIT_REVCOUNT)
	set(GIT_REVCOUNT 0)
endif()
if(NOT GIT_BRANCH)
	set(GIT_BRANCH "codol-1.1.1.13")
endif()

set(VER_PARTS 0 0 0)
if(GIT_TAG MATCHES "([0-9]+)\\.([0-9]+)\\.([0-9]+)")
	set(VER_PARTS ${CMAKE_MATCH_1} ${CMAKE_MATCH_2} ${CMAKE_MATCH_3})
endif()
list(JOIN VER_PARTS "," VER3_RC)
list(JOIN VER_PARTS "." VER3)

set(VERSION_H_CONTENT "/*
 * Automatically generated by CMake.
 * Do not touch!
 */

#define GIT_DESCRIBE \"${GIT_DESCRIBE}\"
#define GIT_DIRTY 0
#define GIT_HASH \"${GIT_HASH}\"
#define GIT_TAG \"${GIT_TAG}\"
#define GIT_BRANCH \"${GIT_BRANCH}\"

// Version transformed for RC files
#define VERSION_PRODUCT_RC ${VER3_RC}
#define VERSION_PRODUCT \"${VER3}\"
#define VERSION_FILE_RC ${VER3_RC},${GIT_REVCOUNT}
#define VERSION_FILE \"${VER3}.${GIT_REVCOUNT}\"

// Alias definitions
#define VERSION GIT_DESCRIBE
#define SHORTVERSION VERSION_PRODUCT
")
set(GEN_DIR ${CMAKE_BINARY_DIR}/generated)
file(CONFIGURE OUTPUT ${GEN_DIR}/version.h CONTENT "${VERSION_H_CONTENT}" @ONLY)
file(CONFIGURE OUTPUT ${GEN_DIR}/version.hpp CONTENT "#include \"version.h\"\n" @ONLY)

# ---------------------------------------------------------------------------
# dependencies (static libs, warnings off)

function(dep_lib name)
	add_library(${name} STATIC ${ARGN})
	target_compile_options(${name} PRIVATE /W0 /MP)
	target_compile_definitions(${name} PRIVATE _WINDOWS WIN32 UNICODE _UNICODE _SILENCE_ALL_CXX17_DEPRECATION_WARNINGS
		$<$<CONFIG:Debug>:DEBUG _DEBUG> $<$<CONFIG:Release>:NDEBUG>)
	set_target_properties(${name} PROPERTIES FOLDER "Dependencies")
endfunction()

# detours (+ bundled zydis)
file(GLOB DETOURS_SRC ${DEPS}/detours/detours/*.cpp ${DEPS}/detours/detours/*.c)
dep_lib(detours ${DETOURS_SRC})
target_include_directories(detours PUBLIC
	${DEPS}/detours
	${DEPS}/detours/detours/zydis/msvc
	${DEPS}/detours/detours/zydis/src
	${DEPS}/detours/detours/zydis/dependencies/zycore/include
	${DEPS}/detours/detours/zydis/include)
target_compile_definitions(detours PUBLIC ZYDIS_STATIC_DEFINE _LIB)
target_compile_options(detours PRIVATE /std:c++latest)

# imgui
file(GLOB IMGUI_SRC ${DEPS}/imgui/*.cpp)
dep_lib(imgui ${IMGUI_SRC}
	${DEPS}/imgui/backends/imgui_impl_dx9.cpp
	${DEPS}/imgui/backends/imgui_impl_win32.cpp
	${DEPS}/imgui/misc/cpp/imgui_stdlib.cpp)
target_include_directories(imgui PUBLIC ${DEPS}/imgui)
target_include_directories(imgui PRIVATE ${DEPS}/dxsdk/Include)
target_compile_options(imgui PRIVATE /std:c++latest)

# ImGuizmo
dep_lib(ImGuizmo ${DEPS}/ImGuizmo/ImGuizmo.cpp)
target_include_directories(ImGuizmo PUBLIC ${DEPS}/ImGuizmo ${DEPS}/imgui)
target_compile_options(ImGuizmo PRIVATE /std:c++latest)

# discord-rpc
file(GLOB DISCORD_SRC ${DEPS}/discord-rpc/src/*.cpp)
list(FILTER DISCORD_SRC EXCLUDE REGEX "(dllmain|_linux|_unix|_osx)\\.cpp$")
dep_lib(discord-rpc ${DISCORD_SRC})
target_include_directories(discord-rpc PUBLIC ${DEPS}/discord-rpc/include)
target_include_directories(discord-rpc PRIVATE ${DEPS}/rapidjson/include)
target_compile_options(discord-rpc PRIVATE /std:c++latest)

# WinHttpClient (built without /std:c++latest, as upstream does)
file(GLOB WINHTTP_SRC ${DEPS}/WinHttpClient/WinHttpClient/*.cpp)
dep_lib(WinHttpClient ${WINHTTP_SRC})
target_include_directories(WinHttpClient PUBLIC ${DEPS}/WinHttpClient/WinHttpClient)

# ---------------------------------------------------------------------------
# iw3r.dll

file(GLOB_RECURSE IW3R_SRC CONFIGURE_DEPENDS
	${CMAKE_SOURCE_DIR}/src/*.cpp
	${CMAKE_SOURCE_DIR}/src/*.hpp
	${CMAKE_SOURCE_DIR}/src/*.rc)

add_library(iw3r SHARED ${IW3R_SRC})
set_target_properties(iw3r PROPERTIES OUTPUT_NAME "iw3r")

target_include_directories(iw3r PRIVATE
	${GEN_DIR}
	${CMAKE_SOURCE_DIR}/src
	${DEPS}/dxsdk/Include
	${DEPS}/fifo_map/src
	${DEPS}/rapidjson/include
	${DEPS}/physx/include
	${DEPS}/physx/include/characterkinematic
	${DEPS}/physx/include/foundation
	${DEPS}/physx/include/common
	${DEPS}/physx/include/extensions)

target_compile_definitions(iw3r PRIVATE _WINDOWS WIN32 UNICODE _UNICODE _SILENCE_ALL_CXX17_DEPRECATION_WARNINGS
	$<$<CONFIG:Debug>:DEBUG _DEBUG> $<$<CONFIG:Release>:NDEBUG>)

target_compile_options(iw3r PRIVATE /std:c++latest /W4 /MP /Zm100
	/wd4239 /wd4505 /wd4996 /wd6001 /wd6385 /wd6386 /wd26812
	$<$<CONFIG:Release>:/Ox /Zi>)
target_link_options(iw3r PRIVATE /SAFESEH:NO $<$<CONFIG:Release>:/DEBUG /OPT:REF>)

target_precompile_headers(iw3r PRIVATE ${CMAKE_SOURCE_DIR}/src/std_include.hpp)
set_source_files_properties(${CMAKE_SOURCE_DIR}/src/Resource.rc PROPERTIES
	COMPILE_FLAGS "/I\"${GEN_DIR}\" /I\"${CMAKE_SOURCE_DIR}/src\"")

target_link_directories(iw3r PRIVATE ${DEPS}/dxsdk/Lib/x86)
set(PHYSX_LIBDIR ${DEPS}/physx/lib/$<IF:$<CONFIG:Debug>,debug,profile>)
target_link_libraries(iw3r PRIVATE
	detours imgui ImGuizmo discord-rpc WinHttpClient
	${PHYSX_LIBDIR}/PhysX_32.lib
	${PHYSX_LIBDIR}/PhysXCharacterKinematic_static_32.lib
	${PHYSX_LIBDIR}/PhysXCommon_32.lib
	${PHYSX_LIBDIR}/PhysXCooking_32.lib
	${PHYSX_LIBDIR}/PhysXExtensions_static_32.lib
	${PHYSX_LIBDIR}/PhysXFoundation_32.lib
	${PHYSX_LIBDIR}/PhysXPvdSDK_static_32.lib)

# PhysX runtime dlls next to the built dll (premake post-build step)
set(PHYSX_BINDIR ${DEPS}/physx/bin/$<IF:$<CONFIG:Debug>,debug,release>)
add_custom_command(TARGET iw3r POST_BUILD
	COMMAND ${CMAKE_COMMAND} -E copy_if_different
		${PHYSX_BINDIR}/PhysX_32.dll ${PHYSX_BINDIR}/PhysXCommon_32.dll
		${PHYSX_BINDIR}/PhysXCooking_32.dll ${PHYSX_BINDIR}/PhysXFoundation_32.dll
		$<TARGET_FILE_DIR:iw3r>)

# Optional: deploy straight into a radiant root (the folder that contains bin/).
set(RADIANT_ROOT "" CACHE PATH "If set, iw3r.dll and the PhysX dlls are copied to <RADIANT_ROOT>/bin after each build")
if(RADIANT_ROOT)
	add_custom_command(TARGET iw3r POST_BUILD
		COMMAND ${CMAKE_COMMAND} -E copy_if_different
			$<TARGET_FILE:iw3r> ${PHYSX_BINDIR}/PhysX_32.dll ${PHYSX_BINDIR}/PhysXCommon_32.dll
			${PHYSX_BINDIR}/PhysXCooking_32.dll ${PHYSX_BINDIR}/PhysXFoundation_32.dll
			${RADIANT_ROOT}/bin)
endif()
